func loadStudySamples(t testing.TB) []savedSample {
	t.Helper()
	path := os.Getenv("FAQT_SAMPLES")
	if path == "" {
		t.Skip("set FAQT_SAMPLES")
	}
	f, err := os.Open(path)
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()
	scan := bufio.NewScanner(f)
	scan.Buffer(make([]byte, 65536), 8000000)
	var samples []savedSample
	for scan.Scan() {
		var s savedSample
		if err := json.Unmarshal(scan.Bytes(), &s); err != nil {
			t.Fatal(err)
		}
		samples = append(samples, s)
	}
	if err := scan.Err(); err != nil {
		t.Fatal(err)
	}
	return samples
}
func TestStudySamples(t *testing.T) {
	samples := loadStudySamples(t)
	ctx := context.Background()
	o := DefaultOptions()
	type counts struct {
		Calls, Extensions, Fallbacks, ScoreDifferences, EndpointDifferences, TraceDifferences int
		Seconds                                                                                                       float64
	}
	results := map[string]counts{}
	for _, name := range []string{"scalar", "fast", "simd", "wfa", "hybrid", "wfa2"} {
		if name == "wfa2" && !studyNativeAvailable {
			continue
		}
		c := counts{}
		start := time.Now()
		for _, s := range samples {
			c.Calls++
			if s.Extend {
				c.Extensions++
			}
			var got dpResult
			var err error
			switch name {
			case "scalar":
				got, err = alignDPReference(ctx, s.Q, s.T, s.Band, s.Extend, o)
			case "fast":
				got, err = alignDPScalar(ctx, s.Q, s.T, s.Band, s.Extend, o)
			case "simd":
				got, err = alignDPPlatform(ctx, s.Q, s.T, s.Band, s.Extend, o)
			case "wfa2":
				got, err = alignWFA2(ctx, s.Q, s.T, s.Band, s.Extend, o)
			default:
				budget := 512
				if name == "hybrid" {
					budget = 64
				}
				var ok bool
				got, ok, err = tryWFA(ctx, s.Q, s.T, s.Band, s.Extend, o, budget)
				if !ok && err == nil {
					c.Fallbacks++
					got, err = alignDPPlatform(ctx, s.Q, s.T, s.Band, s.Extend, o)
				}
			}
			if err != nil {
				t.Fatal(err)
			}
			if got.score != s.Score {
				c.ScoreDifferences++
			}
			if got.q != s.EndQ || got.t != s.EndT {
				c.EndpointDifferences++
			}
			if !reflect.DeepEqual(got.ops, s.Ops) {
				c.TraceDifferences++
			}
		}
		c.Seconds = time.Since(start).Seconds()
		results[name] = c
		t.Log(name, fmt.Sprintf("%+v", c))
		if (name == "scalar" || name == "fast" || name == "simd") && (c.ScoreDifferences+c.EndpointDifferences+c.TraceDifferences > 0) {
			t.Fatalf("%s changed results", name)
		}
	}
	// Time the validation separately from kernel measurements.
	for _, s := range samples {
		nodrop := o
		nodrop.ZDrop = 100000000
		got, ok, err := tryWFA(ctx, s.Q, s.T, s.Band, s.Extend, nodrop, 512)
		if err != nil {
			t.Fatal(err)
		}
		if !ok {
			continue
		}
		want, err := alignDPReference(ctx, s.Q, s.T, s.Band, s.Extend, nodrop)
		if err != nil {
			t.Fatal(err)
		}
		if got.score != want.score || got.q != want.q || got.t != want.t {
			t.Fatalf("WFA disagrees without Z-drop: expected %+v got %+v", want, got)
		}
	}
	if out := os.Getenv("FAQT_SAMPLE_RESULT"); out != "" {
		data, _ := json.MarshalIndent(results, "", "  ")
		if err := os.WriteFile(out, data, 0644); err != nil {
			t.Fatal(err)
		}
	}
}

func BenchmarkKernelStudy(b *testing.B) {
	ctx := context.Background()
	o := DefaultOptions()
	o.ZDrop = 1000000
	for _, n := range []int{32, 128, 256, 1000} {
		q := dna(n, 947)
		targ := slicesCopy(q)
		for i := 15; i < n; i += 37 {
			targ[i] = 'N'
		}
		for name, fn := range map[string]func(context.Context, []byte, []byte, int, bool, Options) (dpResult, error){"scalar": alignDPReference, "fast": alignDPScalar, "simd": alignDPPlatform, "wfa": alignWFA} {
			b.Run(fmt.Sprintf("%d/%s", n, name), func(b *testing.B) {
				b.ReportAllocs()
				for i := 0; i < b.N; i++ {
					_, err := fn(ctx, q, targ, 96, true, o)
					if err != nil {
						b.Fatal(err)
					}
				}
			})
		}
	}
}
