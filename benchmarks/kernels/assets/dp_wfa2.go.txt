import "C"

import (
	"context"
	"slices"
	"sync"
	"unsafe"
)

var nativeMu sync.Mutex
var nativeAligners [2]*C.wavefront_aligner_t

func alignWFA2(ctx context.Context, q, t []byte, band int, extend bool, o Options) (dpResult, error) {
	if err := ctx.Err(); err != nil {
		return dpResult{}, err
	}
	def := DefaultOptions()
	if len(q) == 0 || len(t) == 0 || o.Match != def.Match || o.Mismatch != def.Mismatch || o.GapOpen != def.GapOpen || o.GapExtend != def.GapExtend || o.LongGapOpen != def.LongGapOpen || o.LongGapExtend != def.LongGapExtend {
		return alignDPScalar(ctx, q, t, band, extend, o)
	}
	nativeMu.Lock()
	defer nativeMu.Unlock()
	k := 0
	if extend {
		k = 1
	}
	a := nativeAligners[k]
	if a == nil {
		a = C.study_new(C.int(k))
		nativeAligners[k] = a
	}
	qb, tb := slices.Clone(q), slices.Clone(t)
	for i := range qb {
		if qb[i] == 'N' {
			qb[i] = 'a'
		}
	}
	for i := range tb {
		if tb[i] == 'N' {
			tb[i] = 'b'
		}
	}
	status := C.study_align(a, (*C.char)(unsafe.Pointer(&qb[0])), C.int(len(qb)), (*C.char)(unsafe.Pointer(&tb[0])), C.int(len(tb)), C.int(band))
	if status < 0 {
		return alignDPScalar(ctx, q, t, band, extend, o)
	}
	cigar := a.cigar
	codes := unsafe.Slice((*byte)(unsafe.Pointer(cigar.operations)), int(cigar.end_offset))
	out := dpResult{}
	for pos := int(cigar.begin_offset); pos < int(cigar.end_offset); pos++ {
		code := codes[pos]
		switch code {
		case 'M', 'X', '=':
			code = 'X'
			if exact(q[out.q], t[out.t]) {
				code = '='
			}
			out.q++
			out.t++
		case 'I':
			code = 'D'
			out.t++
		case 'D':
			code = 'I'
			out.q++
		}
		out.ops = addOp(out.ops, code, 1)
		if (out.t == out.q-band && out.t > 0) || (out.t == out.q+band && out.t < len(t)) {
			out.edge = true
		}
	}
	for _, op := range out.ops {
		switch op.Code {
		case '=':
			out.score += op.Length * o.Match
		case 'X':
			out.score -= op.Length * o.Mismatch
		case 'I', 'D':
			out.score -= gapCost(op.Length, o)
		}
	}
	if err := ctx.Err(); err != nil {
		return dpResult{}, err
	}
	return out, nil
}
