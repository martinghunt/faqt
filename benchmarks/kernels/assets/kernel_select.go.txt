package align

import (
	"context"
	"os"
)

var studyKernel = os.Getenv("FAQT_KERNEL")

func alignDP(ctx context.Context, q, t []byte, band int, extend bool, o Options) (dpResult, error) {
	switch studyKernel {
	case "fast":
		return alignDPScalar(ctx, q, t, band, extend, o)
	case "wide":
		return studyWide(ctx, q, t, band, extend, o)
	case "simd":
		return alignDPPlatform(ctx, q, t, band, extend, o)
	case "wfa2":
		if !studyNativeAvailable {
			panic("rebuild with --wfa2 to enable the reference C library")
		}
		return alignWFA2(ctx, q, t, band, extend, o)
	case "wfa":
		return alignWFA(ctx, q, t, band, extend, o)
	case "hybrid":
		r, ok, err := tryWFA(ctx, q, t, band, extend, o, 64)
		if ok || err != nil {
			return r, err
		}
		return alignDPPlatform(ctx, q, t, band, extend, o)
	default:
		r, err := alignDPReference(ctx, q, t, band, extend, o)
		if err == nil {
			captureKernel(q, t, band, extend, r)
		}
		return r, err
	}
}
